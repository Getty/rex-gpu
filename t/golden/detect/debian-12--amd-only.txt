# golden: detect/debian-12--amd-only
# Host interactions Rex::GPU emits, in order. Regenerate with
# REX_GPU_GOLDEN_UPDATE=1 prove -l t/ -- and read the diff.
run: command -v lspci >/dev/null 2>&1
run: lspci -nn 2>&1 | grep -E '\[03(00|02)\]|\[(0380|1200)\].*\[1002:'
run: [ -d /sys/module/amdgpu ] && echo 'amdgpu 1' || echo 'amdgpu 0'; [ -c /dev/kfd ] && echo 'kfd 1' || echo 'kfd 0'; for d in /sys/bus/pci/devices/*; do [ -r "$d/vendor" ] && [ -r "$d/class" ] || continue; read -r v < "$d/vendor"; [ "$v" = 0x1002 ] || continue; read -r c < "$d/class"; case $c in 0x03*|0x1200*) ;; *) continue;; esac; i=-; [ -r "$d/device" ] && read -r i < "$d/device"; r=-; [ -e "$d/driver" ] && r=$(cd -P "$d/driver" 2>/dev/null && echo "${PWD##*/}"); echo "pci ${d##*/} $c $i ${r:--}"; [ -r "$d/mem_info_vram_total" ] && read -r m < "$d/mem_info_vram_total" && echo "vram ${d##*/} $m"; [ -r "$d/mem_info_gtt_total" ] && read -r m < "$d/mem_info_gtt_total" && echo "gtt ${d##*/} $m"; done; for d in /sys/class/drm/renderD*; do [ -r "$d/device/vendor" ] || continue; read -r v < "$d/device/vendor"; s=$(cd -P "$d/device" 2>/dev/null && echo "${PWD##*/}"); echo "render ${d##*/} $v ${s:--}"; done; for n in /sys/class/kfd/kfd/topology/nodes/*; do [ -r "$n/properties" ] || continue; echo "node ${n##*/}"; while read -r k v; do case $k in gfx_target_version|vendor_id|device_id|location_id|domain|drm_render_minor) echo "prop $k $v";; esac; done < "$n/properties"; done; echo end
